 (a) Display OS version, release number, kernel version
 #!/bin/bash
 if [-f /etc/os-release ]
 then
 echo-e "OS: `cat /etc/os-release`"
 fi
 if [-f /etc/shells ]
 then
 echo-e "Available Shells: "
 echo-e "`cat /etc/shells`"
 fi
